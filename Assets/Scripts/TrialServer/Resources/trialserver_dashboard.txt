// The server ACCEPTS a command (202) and runs it on Unity's main thread a frame later. So a 202 means
// "heard you", never "done". The bridge reports the real outcome (accepted or refused) into the status blob,
// and the OUTCOME appears in the message box below the buttons. This line only reports what it truly knows:
// that the headset received the bytes.
function post(path, body) {
  var note = document.getElementById('msgbox');
  return fetch(path, { method: 'POST', body: body === undefined ? '' : body })
    .then(function (r) {
      // The command runs on Unity's next frame. Poll straight away (and again shortly after) rather than
      // waiting up to 500 ms for the next scheduled tick.
      setTimeout(poll, 60);
      setTimeout(poll, 400);
    })
    .catch(function (e) {
      note.className = 'msg error';
      note.textContent = 'Could not reach the headset: ' + e;
    });
}

// The study's SequenceManager takes a 0-based sequence index; the bridge parses the raw body as an int, so
// send the number itself, not JSON.
function selectSequence(i) { post('/api/trial/sequence', String(i)); }

// Arming re-enables the device Start button so the PARTICIPANT starts the trial. This page never starts one
// (/api/trial/start still exists on the server as a bring-up backdoor, deliberately unwired here).
function armTrial() { post('/api/trial/arm'); }

function pauseTrial() { post('/api/trial/pause'); }
function resumeTrial() { post('/api/trial/resume'); }

// Ending early is destructive to the run, so it confirms. It routes through the study's own end path.
function endTrial() {
  if (!confirm('End the current set early? A MANUAL_END marker is written to the gaze JSON.')) return;
  post('/api/trial/end');
}

// The bridge uses the raw body as the marker text, so send the plain string.
function markEvent() {
  var el = document.getElementById('in_mark');
  var text = el.value.trim();
  if (!text) { return; }
  post('/api/trial/mark', text).then(function () { el.value = ''; });
}

// Flag the current (or just-finished) trial invalid; the bridge picks which trial the flag names.
function flagBad() {
  var el = document.getElementById('in_bad');
  post('/api/trial/bad', el.value.trim()).then(function () { el.value = ''; });
}

// The only write the DEV panel makes: start/stop the performance monitor. 202 means "heard you"; the next
// status poll is what confirms it (the perf card's state chip flips).
function togglePerf() {
  var on = document.getElementById('f_perf').classList.contains('on');
  fetch('/api/dev/perf?on=' + (on ? '0' : '1'), { method: 'POST' }).catch(function () {});
}

// Phones lock their screen after ~30 s. For a long session that means constantly waking the phone just to
// see the panel. Wake Lock needs a user gesture, so it is a button. It is dropped whenever the page is
// hidden, hence the re-request on visibilitychange below.
var wakeLock = null;

function setWakeLabel() {
  document.getElementById('wake').textContent = 'Keep this screen awake: ' + (wakeLock ? 'ON' : 'OFF');
}

function toggleWake() {
  if (!('wakeLock' in navigator)) {
    document.getElementById('wake').textContent = 'Screen wake lock not supported by this browser';
    return;
  }
  if (wakeLock) { wakeLock.release(); wakeLock = null; setWakeLabel(); return; }

  navigator.wakeLock.request('screen').then(function (lock) {
    wakeLock = lock;
    lock.addEventListener('release', function () { wakeLock = null; setWakeLabel(); });
    setWakeLabel();
  }).catch(function (e) {
    document.getElementById('wake').textContent = 'Could not keep the screen awake: ' + e.message;
  });
}

    var el = document.getElementById(id);

    if (v === null || v === undefined) return '-';
    return v.toFixed(digits === undefined ? 1 : digits) + (unit || '');
  }

  // null / undefined -> 'n/a'. Deliberately NOT the same helper as n(): a zero that means "this device does
  // not publish that counter" must not look like a zero that means "nothing happened".
  function na(v, dec, unit) { return (v === null || v === undefined) ? 'n/a' : n(v, dec, unit); }

  // One place turns a phase into its display class, so TUTORIAL reads as its own loud state and is never
  // mistaken for a running trial (the tutorial spawns gems and records the moment a sequence is picked).
  var PHASE_CLASS = { Recording: 'rec', Ready: 'ready', Armed: 'armed', Tutorial: 'tut', Paused: 'paused' };
  var PHASE_TEXT = { Tutorial: 'TUTORIAL (not a trial)', Armed: 'ARMED - awaiting participant' };

  function condLabel(t) {
    if (!t.pool) return '-';
    return (t.timeOfDay || '?') + ' + ' + (t.wireframe ? 'Wireframe' : 'No wireframe');
  }

  function mmss(s) {
    var m = Math.floor(s / 60), r = Math.floor(s % 60);
    return m + ':' + (r < 10 ? '0' : '') + r;
  }

  // ---- Health chips + punch list -------------------------------------------------------------------
  // One chip per subsystem, computed from the status blob. Green = fine, amber = a caveat you should know,
  // red = a real problem to fix before recording. Ordered worst-first. Each non-green chip carries a FIX:
  // the plain-language thing to actually go and do, because at 8am with a participant in the chair "restart
  // band-OSC.py" is useful and "EEG: STALE 12s" is not.
  function computeHealth(d) {
    var r = d.ready || {};
    var chips = [];   // {cls, label, fix}, cls in ok|warn|bad

    // Eye tracking permission: the whole game. Denied looks healthy (app runs, UI responds) but the gaze
    // data is empty. Only red once the gaze manager is up; a missing manager is its own separate problem.
    if (!r.gazeManager) chips.push({ cls: 'bad', label: 'Gaze manager: MISSING',
      fix: 'GazeInputManager is not in the scene, so there is no eye ray at all. Check the study prefab is loaded.' });
    else if (!r.eyePermission) chips.push({ cls: 'bad', label: 'Eyes: permission DENIED',
      fix: 'Gems cannot be destroyed by gaze and the recorded gaze data is empty. Accept the eye-tracking popup on the headset (it appears after the photos/videos one) before running anything.' });
    else chips.push({ cls: 'ok', label: 'Eyes: ok' });

    // Tracker present: with it missing, nothing is being written at all.
    if (!r.tracker) chips.push({ cls: 'bad', label: 'Tracker: MISSING',
      fix: 'EyeAndHeadTracker is not in the scene, so nothing is being recorded. Check the study prefab is loaded.' });
    else chips.push({ cls: 'ok', label: 'Tracker: ok' });

    // Head tracking. Tri-state: null off-device (no HMD, e.g. the Editor) so no chip at all; on-device, red
    // when the head pose is lost. This is the OUTDOOR failure: a featureless sky drops tracking and every
    // head-relative number silently goes wrong while the app still looks alive.
    if (r.headTracked === false) chips.push({ cls: 'bad', label: 'Head tracking: LOST',
      fix: 'The headset has lost its position (common outdoors against open sky or blank walls). Head-relative data is corrupt until it recovers. Give it visual features to lock onto, or pause until the pose returns.' });
    else if (r.headTracked === true) chips.push({ cls: 'ok', label: 'Head tracking: ok' });

    // Battery, if the platform reports it.
    var vt = d.vitals || {};
    var bp = vt.batteryPct;
    if (bp !== null && bp !== undefined) {
      if (bp < 15 && !vt.charging) chips.push({ cls: 'bad', label: 'Battery: ' + n(bp, 0) + '%',
        fix: 'If the headset dies mid-trial that run is cut off. Finish the current trial and charge.' });
      else chips.push({ cls: 'ok', label: 'Battery: ' + n(bp, 0) + '%' + (vt.charging ? ' (chg)' : '') });
    }

    // Performance. Only speaks when the monitor is actually sampling: with it off there is nothing honest to
    // say, and a chip that says "perf: unknown" on every run is one everyone learns to skip. These are the
    // chips that matter for the crash/overheating problem this whole card exists to catch.
    var p = (d.dev && d.dev.perf) || {};
    if (p.on) {
      if (p.freeMb !== null && p.freeMb !== undefined && p.freeCritMb && p.freeMb < p.freeCritMb) {
        chips.push({ cls: 'bad', label: 'Storage: ' + n(p.freeMb, 0) + ' MB free',
          fix: 'The headset is nearly out of space. A gaze session file will stop mid-write, silently. Download and clear old files before recording anything else.' });
      }
      if (p.thermal !== null && p.thermal !== undefined && p.thermalWarn && p.thermal >= p.thermalWarn) {
        chips.push({ cls: 'bad', label: 'Thermal: throttling (' + p.thermal + ')',
          fix: 'The headset is hot enough to be throttling, so slow frames right now are the HARDWARE, not a code regression. Let it cool before drawing any conclusion, and before a long session.' });
      }
      if (p.overPct !== null && p.overPct !== undefined && p.overPct > (p.overBudgetWarnPct || 20)) {
        chips.push({ cls: 'warn', label: 'Frames: ' + n(p.overPct, 0) + '% over budget',
          fix: 'A lot of frames are missing the budget (1% low is ' + na(p.lowFps, 1) + ' fps). Check the thermal state first, then how much is on screen at once.' });
      } else {
        chips.push({ cls: 'ok', label: 'Perf: ' + na(p.meanFps, 0) + ' fps' });
      }
    }

    // CRIT log lines. A subsystem has already declared this session's data compromised, and without this the
    // only way to find out is to open DEV and read the log. Not a per-subsystem chip: the catch-all for
    // "something said CRIT".
    if (d.crit && d.crit.count > 0) {
      chips.push({ cls: 'bad', label: 'Log: ' + d.crit.count + ' CRITICAL',
        fix: 'Latest: ' + (d.crit.latest || '(no message)') + '  -  open DEV, filter the log to CRIT for the full history.' });
    }

    var order = { bad: 0, warn: 1, ok: 2 };
    chips.sort(function (a, b) { return order[a.cls] - order[b.cls]; });
    return chips;
  }

  function renderHealth(chips) {
    document.getElementById('healthbar').innerHTML = chips.map(function (c) {
      return '<span class="chip ' + c.cls + '">' + esc(c.label) + '</span>';
    }).join('');
  }

  // The run-day punch list: everything that needs fixing, in one card, worst first. An EMPTY list is the
  // whole point, so the empty state is a loud green "nothing needs fixing" rather than a blank box that reads
  // the same as "not loaded yet".
  function renderPunchList(chips, d) {
    var bad = chips.filter(function (c) { return c.cls !== 'ok'; });
    var box = document.getElementById('punchlist');
    var card = document.getElementById('punchcard');

    var reds = bad.filter(function (c) { return c.cls === 'bad'; }).length;
    card.classList.toggle('clear', bad.length === 0);
    card.classList.toggle('hasred', reds > 0);

    if (bad.length === 0) {
      box.innerHTML = '<div class="pl clear">Nothing needs fixing. Every check is green.</div>';
      return;
    }

    box.innerHTML = bad.map(function (c) {
      return '<div class="pl ' + c.cls + '">' +
             '<div class="pltitle">' + esc(c.label) + '</div>' +
             (c.fix ? '<div class="plfix">' + esc(c.fix) + '</div>' : '') +
             '</div>';
    }).join('');
  }

  // ---- Performance ---------------------------------------------------------------------------------
  var perfHistory = [];       // client-side, purely for the sparkline
  var PERF_HISTORY = 120;     // about a minute at the 2 Hz poll

  function renderPerf(p) {
    var on = !!p.on;
    var btn = document.getElementById('f_perf');
    btn.textContent = on ? 'Turn sampling OFF' : 'Turn sampling ON';
    btn.classList.toggle('on', on);

    // Frame rate is ALWAYS live (PerfMonitor lerps it whether or not sampling is on).
    txt('d_fps', na(p.fps, 1) + ' fps  (' + na(p.frameMs, 1, ' ms') + ')');

    document.getElementById('d_perfstate').innerHTML = on
      ? '<span class="dot ok"></span>sampling ' + na(p.forS, 0, 's')
      : '<span class="dot"></span>off';

    if (!on) {
      perfHistory = [];
      drawSpark();
      ['d_pfps','d_ppct','d_pworst','d_pover','d_pmem','d_pgc','d_pdraw','d_pgpu','d_pdrop','d_ptherm','d_pbatt','d_pfree','d_psub']
        .forEach(function (id) { txt(id, '-'); });
      txt('d_pnote', 'Sampling is off, the shipped default: telemetry costs frame time and a recorded trial ' +
                     'should not pay for it. Turn it on when you are asking a performance question, and off ' +
                     'again before you record. The frame rate above is always live.');
      return;
    }

    perfHistory.push(p.p95 === undefined ? null : p.p95);
    if (perfHistory.length > PERF_HISTORY) perfHistory.shift();
    drawSpark(p.budgetMs);

    txt('d_pfps', na(p.meanFps, 1) + ' / ' + na(p.lowFps, 1) + ' fps');
    txt('d_ppct', na(p.p50, 1) + ' / ' + na(p.p95, 1) + ' / ' + na(p.p99, 1) + ' ms');
    txt('d_pworst', na(p.worstMs, 1, ' ms') + '  (' + na(p.samples, 0) + ' frames in window)');
    txt('d_pover', na(p.overPct, 1, '%') + ' over ' + na(p.budgetMs, 1) + ' ms  /  ' + na(p.hitches, 0) + ' hitch(es)');
    txt('d_pmem', na(p.allocMb, 0, ' MB') + ' / ' + na(p.rssMb, 0, ' MB') + ' / ' + na(p.rssPeakMb, 0, ' MB'));
    txt('d_pgc', p.gcPerFrame === null ? 'n/a (needs a development build)' : n(p.gcPerFrame / 1024, 1, ' KB'));
    txt('d_pdraw', p.drawCalls === null ? 'n/a (needs a development build)' : na(p.drawCalls, 0) + ' / ' + na(p.setPass, 0));
    txt('d_pgpu', na(p.gpuAppMs, 2, ' ms') + ' / ' + na(p.gpuCompMs, 2, ' ms'));
    txt('d_pdrop', na(p.dropped, 0) + (p.presented === null ? '' : ' of ' + n(p.presented, 0) + ' presented'));

    var TH = ['NONE','LIGHT','MODERATE','SEVERE','CRITICAL','EMERGENCY','SHUTDOWN'];
    txt('d_ptherm', p.thermal === null ? 'n/a (needs Android API 29+)'
      : (TH[p.thermal] || p.thermal) + ' / ' + (TH[p.thermalPeak] || p.thermalPeak));
    txt('d_pbatt', na(p.batteryPct, 0, '%') +
      (p.drainPctPerHour === null ? '  (rate not established yet)'
                                  : '  -' + n(p.drainPctPerHour, 1) + '%/h, about ' + na(p.hoursLeft, 1) + ' h left'));
    txt('d_pfree', na(p.freeMb, 0, ' MB'));

    var subs = p.subsystems || {};
    var sk = Object.keys(subs);
    txt('d_psub', sk.length ? sk.map(function (k) { return k + ' ' + n(subs[k], 2) + ' ms'; }).join(', ')
                            : 'none reporting (call PerfMonitor.ReportSubsystemMs from your frame hogs)');

    var note = [];
    if (!p.devBuild) note.push('This is not a development build, so draw calls, SetPass and GC-per-frame read n/a. Everything else is real.');
    if (!p.xrStats) note.push('The XR runtime is not publishing compositor stats on this device, so GPU times and dropped frames read n/a. App frame rate alone can look healthy while the compositor is reprojecting.');
    txt('d_pnote', note.join(' '));
  }

  // A bare frame-time sparkline: p95 over the last minute, with the budget drawn as a line across it. The
  // shape is the point. A flat trace at the budget is a healthy session; a rising one is thermal throttling
  // creeping in, visible here long before it is obvious in any single number.
  function drawSpark(budgetMs) {
    var c = document.getElementById('perfspark');
    if (!c) return;
    var w = c.clientWidth || 300, h = c.height;
    if (c.width !== w) c.width = w;
    var g = c.getContext('2d');
    g.clearRect(0, 0, w, h);
    if (perfHistory.length < 2) return;

    var max = 0;
    perfHistory.forEach(function (v) { if (v !== null && v > max) max = v; });
    if (budgetMs && budgetMs > max) max = budgetMs;
    max = max * 1.15 || 1;

    if (budgetMs) {
      var by = h - (budgetMs / max) * h;
      g.strokeStyle = '#4b5560'; g.setLineDash([4, 4]); g.beginPath();
      g.moveTo(0, by); g.lineTo(w, by); g.stroke(); g.setLineDash([]);
    }

    g.strokeStyle = '#6fc3ff'; g.lineWidth = 1.5; g.beginPath();
    var step = w / (PERF_HISTORY - 1);
    var started = false;
    perfHistory.forEach(function (v, i) {
      if (v === null) return;
      var x = i * step, y = h - (v / max) * h;
      if (!started) { g.moveTo(x, y); started = true; } else g.lineTo(x, y);
    });
    g.stroke();
  }

  // ---- DEV MODE ------------------------------------------------------------------------------------
  // Read-only. On run day, with no PC tethered to the headset, the log stream is how you answer "did the
  // bridge bind to the study, and why was that command refused" from the phone already in your hand.

  var devOn = localStorage.getItem('dev') === '1';

  window.toggleDev = function () {
    devOn = !devOn;
    localStorage.setItem('dev', devOn ? '1' : '0');
    applyDev();
    if (devOn) { pollLogs(); pollFiles(true); }
  };

  function applyDev() {
    document.getElementById('devPanel').classList.toggle('hidden', !devOn);
    var b = document.getElementById('devbtn');
    b.textContent = 'DEV: ' + (devOn ? 'ON' : 'OFF');
    b.classList.toggle('on', devOn);
  }

  // ---- Log stream ----------------------------------------------------------------------------------
  // Cursor-based: ask for everything since the last sequence number seen, so lines are never fetched twice
  // and never skipped. If the headset ring wrapped while we were away it reports how many lines dropped, and
  // we say so rather than presenting a log with a silent hole in it. Tags are derived from the lines
  // themselves, so the filter buttons are always whatever the app is actually logging. Severity is filtered
  // SERVER-side (like the tag), so the cursor and the Download button agree with what is on screen.

  var logCursor = 0;
  var logTag = '';          // '' = all subsystems
  var logLevel = '';        // '' = every severity; otherwise a FLOOR ("WARN" shows WARN, ERROR and CRIT)
  var logFollow = true;
  var logLines = [];        // held client-side so the text filter can re-apply without refetching
  var MAX_HELD = 1500;
  var knownTags = [];

  window.toggleFollow = function () {
    logFollow = !logFollow;
    document.getElementById('f_follow').classList.toggle('on', logFollow);
  };

  window.clearLog = function () { logLines = []; drawLog(); };

  window.setLogTag = function (tg) {
    logTag = tg;
    // The server filters by tag, so switching tags means re-reading the ring from the start rather than
    // continuing the cursor: otherwise only NEW lines of that tag would show and the history would look
    // empty, which reads as "this subsystem is silent" when it is not.
    logCursor = 0;
    logLines = [];
    drawTagFilters();
    pollLogs();
  };

  // Same reasoning as the tag: severity is filtered server-side, so switching it re-reads from the start.
  // CRIT gets its own button rather than making you remember it is included in "ERROR and worse".
  window.setLogLevel = function (l) {
    logLevel = l;
    logCursor = 0;
    logLines = [];
    drawTagFilters();
    pollLogs();
  };

  function drawTagFilters() {
    var levels = [['', 'ALL LEVELS'], ['WARN', 'WARN+'], ['ERROR', 'ERROR+'], ['CRIT', 'CRIT']];
    var html = levels.map(function (p) {
      return '<button class="f lv' + (logLevel === p[0] ? ' on' : '') +
             '" onclick="setLogLevel(\'' + p[0] + '\')">' + p[1] + '</button>';
    }).join('') + '<span class="fsep"></span>';

    html += '<button class="f' + (logTag === '' ? ' on' : '') + '" onclick="setLogTag(\'\')">ALL</button>';
    knownTags.forEach(function (tg) {
      html += '<button class="f' + (logTag === tg ? ' on' : '') + '" onclick="setLogTag(\'' + tg + '\')">' + tg + '</button>';
    });
    document.getElementById('logfilters').innerHTML = html;
  }

  function pollLogs() {
    if (!devOn) return;

    fetch('/api/logs?max=300&since=' + logCursor +
          (logTag ? '&tag=' + encodeURIComponent(logTag) : '') +
          (logLevel ? '&level=' + encodeURIComponent(logLevel) : ''),
          { cache: 'no-store' })
      .then(function (r) { return r.json(); })
      .then(function (d) {
        logCursor = d.cursor;

        if (d.dropped > 0) {
          logLines.push({ seq: -1, level: 'WARN', tag: '', t: 0,
                          msg: '... ' + d.dropped + ' line(s) dropped: the headset ring buffer wrapped before this page read them.' });
        }
        if (d.lines.length) {
          logLines = logLines.concat(d.lines);
          // Derive tag filter buttons from what has actually been logged.
          var changed = false;
          d.lines.forEach(function (l) {
            if (l.tag && knownTags.indexOf(l.tag) < 0) { knownTags.push(l.tag); changed = true; }
          });
          if (changed) { knownTags.sort(); drawTagFilters(); }
        }
        if (logLines.length > MAX_HELD) logLines = logLines.slice(logLines.length - MAX_HELD);

        if (d.lines.length || d.dropped) drawLog();
      })
      .catch(function () {});
  }

  function drawLog() {
    var box = document.getElementById('logbox');
    var q = document.getElementById('logsearch').value.trim().toLowerCase();

    var html = logLines.filter(function (l) {
      return !q || l.msg.toLowerCase().indexOf(q) >= 0 || (l.tag || '').toLowerCase().indexOf(q) >= 0;
    }).map(function (l) {
      var lvl = (l.level || 'INFO').toLowerCase();
      return '<div class="ln ' + lvl + '"><span class="t">' + l.t.toFixed(1) + 's</span> ' +
             (lvl === 'info' ? '' : '<span class="lv ' + lvl + '">' + esc(l.level) + '</span> ') +
             (l.tag ? '<span class="tg">[' + esc(l.tag) + ']</span> ' : '') + esc(l.msg) + '</div>';
    }).join('');

    box.innerHTML = html || '<div class="ln"><span class="t">no lines match.</span></div>';
    if (logFollow) box.scrollTop = box.scrollHeight;
  }

  // Pull the headset's whole log ring to a .txt without adb, respecting BOTH current filters (tag and level),
  // so what you save is what you were looking at. Navigate to the endpoint; the browser saves it.
  window.downloadLog = function () {
    var q = [];
    if (logTag) q.push('tag=' + encodeURIComponent(logTag));
    if (logLevel) q.push('level=' + encodeURIComponent(logLevel));
    window.location = '/api/logs/download' + (q.length ? '?' + q.join('&') : '');
  };

  function esc(s) {
    return String(s).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
  }

  // ---- Session files -------------------------------------------------------------------------------
  // Listing recurses the app's persistent data directory ON THE HEADSET. Doing that twice a second is real disk
  // work for a list that changes once per trial. Every 5 s, redrawn only when it changed, and only while DEV
  // is open (the list lives in the DEV panel).

  var lastFilesJson = '';
  var filesTimer = 0;

  function pollFiles(force) {
    if (!devOn) return;
    if (!force && Date.now() - filesTimer < 5000) return;
    filesTimer = Date.now();

    fetch('/api/files', { cache: 'no-store' })
      .then(function (r) { return r.text(); })
      .then(function (text) {
        if (text === lastFilesJson) return;
        lastFilesJson = text;
        var d = JSON.parse(text);

        var ul = document.getElementById('files');
        if (!d.files.length) { ul.innerHTML = '<li class="k">none yet</li>'; return; }

        ul.innerHTML = d.files.map(function (f) {
          return '<li><a href="/api/file?name=' + encodeURIComponent(f.path) + '">' + esc(f.name) + '</a>' +
                 ' <span class="k">(' + (f.bytes / 1024).toFixed(0) + ' KB)</span></li>';
        }).join('');
      })
      .catch(function () {});
  }

  function poll() {
    fetch('/api/status', { cache: 'no-store' })
      .then(function (r) { return r.json(); })
      .then(function (d) { lastOk = Date.now(); render(d); })
      .catch(function () { /* handled by the staleness check */ });

    // Only when DEV is open. With it closed the page costs exactly what it always did, so the run-day
    // dashboard is not paying for a log stream nobody is reading.
    if (devOn) { pollLogs(); pollFiles(false); }
  }

  // Staleness is checked on the CLIENT clock, not the server's. A dashboard that silently keeps showing the
  // last good numbers after the headset drops is worse than one that shows nothing.
  setInterval(function () {
    var age = (Date.now() - lastOk) / 1000;
    var stale = lastOk > 0 && age > 3;
    document.getElementById('staleBox').classList.toggle('hidden', !stale);
    if (stale) txt('staleAge', age.toFixed(0));
    if (lastOk === 0) txt('conn', 'no response from the headset yet');
  }, 1000);

  // Phones freeze background JS timers when the screen locks or you switch apps. On return, lastOk is minutes
  // old and the staleness check would scream that the headset died when nothing happened. A false alarm
  // mid-session is the alarm you learn to ignore.
  document.addEventListener('visibilitychange', function () {
    if (document.visibilityState !== 'visible') return;
    lastOk = Date.now();                                  // do not judge the link on a timer that was frozen
    document.getElementById('staleBox').classList.add('hidden');
    filesTimer = 0;
    poll();                                               // re-establish the truth immediately

    // The browser silently drops the wake lock while hidden, so it has to be retaken or the screen starts
    // sleeping again halfway through the session with the button still reading ON.
    if (wakeLock === null && document.getElementById('wake').textContent.indexOf('ON') >= 0) toggleWake();
  });

  window.poll = poll;   // so a button press can refresh immediately instead of waiting for the next tick

  // The text filter redraws from what is already held, so typing never refetches.
  document.getElementById('logsearch').addEventListener('input', drawLog);

  applyDev();           // DEV survives a reload, so a debugging session is not undone by a refresh
  drawTagFilters();     // seed the level buttons before any log arrives
  if (devOn) { pollLogs(); pollFiles(true); }

  poll();
  setInterval(poll, 500);   // 2 Hz. Nothing here needs to be faster; the data path is not this page.
})();
</script>
