<script>
// The server ACCEPTS a command (202) and runs it on Unity's main thread a frame later. So a 202 means
// "heard you", never "done".
//
// This used to be the whole story, and it was a trap: a refused command still returned 202, this line still
// said "Sent", and the actual reason ("no participant set", "eyes never calibrated") went to logcat on a
// headset with no PC attached to it. The operator saw a confirmation and watched nothing happen. Now the
// controller reports every outcome into the status blob, so this line reports only what it actually knows
// (the headset received the bytes) and the OUTCOME appears in the message box below the buttons.
function post(path, body) {
  var note = document.getElementById('cmdnote');
  note.textContent = 'Sending ' + path + '...';
  return fetch(path, { method: 'POST', body: body || '' })
    .then(function (r) {
      note.textContent = r.ok ? 'The headset received ' + path + '. The result appears below.'
                              : 'Server returned ' + r.status + ' for ' + path;
      // The command runs on Unity's next frame. Poll straight away (and again shortly after) rather than
      // waiting up to 500 ms for the next scheduled tick: the button felt dead when the command had in fact
      // already run and only the display was behind.
      setTimeout(poll, 60);
      setTimeout(poll, 400);
    })
    .catch(function (e) { note.textContent = 'Could not reach the headset: ' + e; });
}

function cmd(name) { post('/api/trial/' + name); }

// Phones lock their screen after ~30 s. For a long walking session that means constantly waking the phone
// just to see the panel. Wake Lock needs a user gesture, so it is a button, not automatic. It is also
// dropped by the browser whenever the page is hidden, hence the re-request on visibilitychange below.
var wakeLock = null;

function setWakeLabel() {
  document.getElementById('wake').textContent = 'Keep this screen awake: ' + (wakeLock ? 'ON' : 'OFF');
}

function toggleWake() {
  if (!('wakeLock' in navigator)) {
    document.getElementById('wake').textContent = 'Screen wake lock not supported by this browser';
    return;
  }
  if (wakeLock) { wakeLock.release(); wakeLock = null; setWakeLabel(); return; }

  navigator.wakeLock.request('screen').then(function (lock) {
    wakeLock = lock;
    lock.addEventListener('release', function () { wakeLock = null; setWakeLabel(); });
    setWakeLabel();
  }).catch(function (e) {
    document.getElementById('wake').textContent = 'Could not keep the screen awake: ' + e.message;
  });
}

// Stopping is the one place a run-day mistake is expensive, so the verdict is explicit: there is no plain
// "Stop" that guesses for you. GOOD files it under valid/, BAD under invalid/ with your reason and retakes
// the trial number. Both keep the file.
function stopTrial(valid) {
  var note = document.getElementById('in_note').value.trim();
  if (!valid && !note && !confirm('Mark this trial BAD with no reason given?')) return;
  post('/api/trial/stop', JSON.stringify({ valid: valid, note: note }))
    .then(function () { document.getElementById('in_note').value = ''; });
}

// The trial number is optional here: send it only when the box has something in it, because a 0 would be
// ignored by the controller anyway and an empty box must mean "leave the counter alone", not "reset it".
function identify() {
  var p = document.getElementById('in_participant').value.trim();
  var c = document.getElementById('in_condition').value.trim();
  var t = parseInt(document.getElementById('in_trial').value, 10);
  if (!p || !c) { document.getElementById('cmdnote').textContent = 'Enter both a participant and a condition.'; return; }

  var body = { participant: p, condition: c };
  if (t > 0) body.trial = t;
  post('/api/trial/identify', JSON.stringify(body));
}

// Abandon the running trial (files it invalid and flags the next start as a redo), or, from Idle, just arm
// the redo flag for the next start.
function redoTrial() {
  var note = document.getElementById('in_note').value.trim() || 'redo';
  post('/api/trial/redo', JSON.stringify({ note: note }))
    .then(function () { document.getElementById('in_note').value = ''; });
}

// Stamp an event into the running CSV without stopping. The trial keeps recording; the moment is now in the
// data instead of in someone's memory.
function markEvent() {
  var el = document.getElementById('in_mark');
  var text = el.value.trim();
  if (!text) { document.getElementById('cmdnote').textContent = 'Type what happened, then press Mark event.'; return; }
  post('/api/trial/mark', JSON.stringify({ note: text }))
    .then(function () { el.value = ''; });
}

    el.innerHTML = '<span class="dot ' + (good ? 'ok' : '') + '"></span>' + (good ? goodText : badText);
  }

  function n(v, digits, unit) {
    if (v === null || v === undefined) return '-';
    return v.toFixed(digits === undefined ? 1 : digits) + (unit || '');
  }

  function render(d) {
    var st = d.trial.state;
    var el = document.getElementById('state');
    el.textContent = st === 'Recording' ? 'RECORDING' : st.toUpperCase();
    el.className = 'state v ' + (st === 'Recording' ? 'rec' : st === 'Paused' ? 'paused' : 'idle');

    txt('participant', d.trial.participant || '(not set)');
    txt('condition', d.trial.condition || '(not set)');
    txt('number', d.trial.number + (d.trial.redoPending ? ' (redo pending)' : ''));
    txt('file', d.trial.file || '-');
    txt('elapsed', n(d.trial.elapsed, 1, ' s'));
    txt('rows', d.trial.rows);

    // The outcome of the last command, refusals included. It ages out after 20 s: a reason still sitting
    // under a button you pressed ten minutes ago reads as a fresh refusal of the one you just pressed.
    var msg = document.getElementById('msgbox');
    var showMsg = d.trial.message && d.trial.messageAge < 20;
    msg.className = 'msg ' + (showMsg ? (d.trial.messageLevel || 'ok') : 'hidden');
    if (showMsg) msg.textContent = d.trial.message;

    // ---- Vitals. Loud, because these are the failures nothing else on the page would reveal. ----
    var vt = d.vitals || {};

    if (vt.batteryPct === null || vt.batteryPct === undefined) {
      txt('v_batt', 'not reported');
    } else {
      var lowBatt = vt.batteryPct < 15 && !vt.charging;
      document.getElementById('v_batt').innerHTML =
        '<span class="dot ' + (vt.batteryPct < 15 ? '' : vt.batteryPct < 30 ? 'warn' : 'ok') + '"></span>' +
        vt.batteryPct.toFixed(0) + '%' + (vt.charging ? ' (charging)' : '');
      document.getElementById('battBox').classList.toggle('hidden', !lowBatt);
      txt('battPct', vt.batteryPct.toFixed(0));
    }

    // Only shout when there IS a headset to lose tracking on. With no XR device (the Editor, a browser test)
    // "not tracked" is a different situation entirely, and a red banner on every desk test is how an alarm
    // gets ignored on the day it matters.
    if (!vt.xrPresent) {
      document.getElementById('v_track').innerHTML = '<span class="dot warn"></span>no XR device';
      document.getElementById('trackBox').classList.add('hidden');
    } else {
      flag('v_track', vt.headTracked, 'tracking', 'LOST for ' + n(vt.trackingLostFor, 1, ' s'));
      document.getElementById('trackBox').classList.toggle('hidden', !vt.trackingLost);
    }

    var r = d.ready;
    flag('r_eye', r.eyeTracking, 'active', 'not tracking');
    flag('r_eyeperm', r.eyePermission, 'granted', 'denied');
    flag('r_pupilperm', r.pupilPermission, 'granted', 'denied');
    flag('r_calib', r.sawHighConfidence, 'yes', 'NOT SEEN');

    // Only a real warning when the reposition is actually in use. Off means you are starting trials from
    // wherever the aircraft sits, which is a choice, not a fault.
    if (!r.useTrialReset) {
      document.getElementById('r_pose').innerHTML = '<span class="dot warn"></span>reset off (manual setup)';
    } else {
      flag('r_pose', r.startPoseCaptured, 'captured', 'NOT CAPTURED');
    }

    document.getElementById('calib').classList.toggle('hidden', !r.calibrationSuspect);

    var chips = '';
    r.seen.forEach(function (k) { chips += '<span class="chip ok">' + k + ' seen</span>'; });
    r.unseen.forEach(function (k) { chips += '<span class="chip bad">' + k + ' NOT seen</span>'; });
    document.getElementById('seen').innerHTML = chips || '<span class="k">no gaze targets in the scene yet</span>';

    var e = d.eye;
    txt('e_conf', e.confidence + (e.confidence >= 2 ? ' (high)' : e.confidence == 1 ? ' (medium)' : ' (low)'));
    txt('e_aoi', d.aoi.current || '(nothing)');
    txt('e_dwell', n(d.aoi.dwell, 1, ' s'));
    txt('e_behavior', e.behavior || '-');
    txt('e_pupil', n(e.leftPupilMm, 2) + ' / ' + n(e.rightPupilMm, 2) + ' mm');
    txt('e_open', n(e.leftOpenness, 2) + ' / ' + n(e.rightOpenness, 2));

    var ar = d.wideAreaAR;
    if (ar) {
      txt('ar_time', n(ar.timeElapsed, 1, ' s'));
      txt('ar_destroyed', String(ar.destroyedCount));
      var tHtml = (ar.remainingTargets || []).map(function(t) {
        return 'X: ' + n(t.x, 2) + ' Y: ' + n(t.y, 2) + ' Z: ' + n(t.z, 2);
      }).join('<br>');
      document.getElementById('ar_targets').innerHTML = tHtml || '(none)';
    } else {
      txt('ar_time', '-');
      txt('ar_destroyed', '-');
      document.getElementById('ar_targets').innerHTML = '-';
    }

    txt('conn', 'headset ' + (d.ip || '(usb tether)') + ':' + d.port + '  |  updated ' + new Date().toLocaleTimeString());

    if (devOn) renderDev(d);
  }

  // ---- DEV MODE ------------------------------------------------------------------------------------
  // Read-only. It exists so that on run day, with no PC tethered to the headset, the questions that used to
  // need `adb logcat` (what is the write path doing, is the receiver actually connected or just in test
  // mode, what is the frame rate, which knob got changed) are answerable from the phone already in your hand.

  var devOn = localStorage.getItem('dev') === '1';
  var devPane = 'log';

  window.toggleDev = function () {
    devOn = !devOn;
    localStorage.setItem('dev', devOn ? '1' : '0');
    applyDev();
    if (devOn) { pollTunables(true); pollLogs(); }
  };

  function applyDev() {
    document.getElementById('devPanel').classList.toggle('hidden', !devOn);
    var b = document.getElementById('devbtn');
    b.textContent = 'DEV: ' + (devOn ? 'ON' : 'OFF');
    b.classList.toggle('on', devOn);
  }

  window.devTab = function (name) {
    devPane = name;
    ['log', 'sys', 'tun'].forEach(function (p) {
      document.getElementById('pane_' + p).classList.toggle('hidden', p !== name);
      document.getElementById('tab_' + p).classList.toggle('on', p === name);
    });
    if (name === 'tun') pollTunables(true);
  };

  function renderDev(d) {
    var v = d.dev;
    if (!v) return;

    txt('d_fps', n(v.fps, 1));
    txt('d_ms', n(v.frameMs, 1, ' ms'));
    txt('d_loglines', v.logLines);
    txt('d_tuncount', v.tunables.count + ' (' + v.tunables.overridden + ' changed)');



    var totals = d.aoi.totals || {};
    var keys = Object.keys(totals);
    document.getElementById('d_dwell').innerHTML = keys.length
      ? keys.map(function (k) {
          return '<div class="row"><span class="k">' + k + '</span><span class="v">' + n(totals[k], 1, ' s') + '</span></div>';
        }).join('')
      : '<span class="k">nothing yet</span>';
  }

  // ---- Log stream ----------------------------------------------------------------------------------
  // Cursor-based: we ask for everything since the last sequence number we saw, so lines are never fetched
  // twice and never skipped. If the headset's ring wrapped while we were away it tells us how many lines
  // were dropped, and we say so, rather than presenting a log with a silent hole in it.

  var logCursor = 0;
  var logTag = '';          // '' = all subsystems
  var logFollow = true;
  var logLines = [];        // held client-side so the text filter can re-apply without refetching
  var MAX_HELD = 1500;
  var knownTags = [];
  var logTimer = 0;

  window.toggleFollow = function () {
    logFollow = !logFollow;
    document.getElementById('f_follow').classList.toggle('on', logFollow);
  };

  window.clearLog = function () {
    logLines = [];
    drawLog();
  };

  window.setLogTag = function (t) {
    logTag = t;
    // The server filters by tag, so switching tags means re-reading the ring from the start rather than
    // continuing the cursor: otherwise you would see only NEW lines of that tag and the history would look
    // empty, which reads as "this subsystem is silent" when it is not.
    logCursor = 0;
    logLines = [];
    drawTagFilters();
    pollLogs();
  };

  function drawTagFilters() {
    var html = '<button class="f' + (logTag === '' ? ' on' : '') + '" onclick="setLogTag(\'\')">ALL</button>';
    knownTags.forEach(function (t) {
      html += '<button class="f' + (logTag === t ? ' on' : '') + '" onclick="setLogTag(\'' + t + '\')">' + t + '</button>';
    });
    document.getElementById('logfilters').innerHTML = html;
  }

  function pollLogs() {
    if (!devOn) return;

    fetch('/api/logs?max=300&since=' + logCursor + (logTag ? '&tag=' + encodeURIComponent(logTag) : ''),
          { cache: 'no-store' })
      .then(function (r) { return r.json(); })
      .then(function (d) {
        logCursor = d.cursor;

        if (d.dropped > 0) {
          logLines.push({ seq: -1, type: 'warn', tag: '', t: 0,
                          msg: '... ' + d.dropped + ' line(s) dropped: the headset ring buffer wrapped before this page read them.' });
        }
        if (d.lines.length) logLines = logLines.concat(d.lines);
        if (logLines.length > MAX_HELD) logLines = logLines.slice(logLines.length - MAX_HELD);

        if (d.lines.length || d.dropped) drawLog();
        txt('lognote', d.total + ' lines logged on the headset since launch. Showing ' + logLines.length + '.');
      })
      .catch(function () {});

    // The tag list only grows, and only when a subsystem logs for the first time. Every 5 s is plenty.
    if (Date.now() - logTimer > 5000) {
      logTimer = Date.now();
      fetch('/api/logtags', { cache: 'no-store' })
        .then(function (r) { return r.json(); })
        .then(function (d) {
          if (d.tags.join(',') === knownTags.join(',')) return;
          knownTags = d.tags;
          drawTagFilters();
        })
        .catch(function () {});
    }
  }

  function drawLog() {
    var box = document.getElementById('logbox');
    var q = document.getElementById('logsearch').value.trim().toLowerCase();

    var html = logLines.filter(function (l) {
      return !q || l.msg.toLowerCase().indexOf(q) >= 0 || (l.tag || '').toLowerCase().indexOf(q) >= 0;
    }).map(function (l) {
      return '<div class="ln ' + l.type + '"><span class="t">' + l.t.toFixed(1) + 's</span> ' +
             (l.tag ? '<span class="tg">[' + esc(l.tag) + ']</span> ' : '') + esc(l.msg) + '</div>';
    }).join('');

    box.innerHTML = html || '<div class="ln"><span class="t">no lines match.</span></div>';
    if (logFollow) box.scrollTop = box.scrollHeight;
  }

  function esc(s) {
    return String(s).replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
  }

  // ---- Tunables ------------------------------------------------------------------------------------

  var tunables = [];
  var tunTimer = 0;
  var overridesOnly = false;

  window.toggleOverridesOnly = function () {
    overridesOnly = !overridesOnly;
    document.getElementById('f_ovr').classList.toggle('on', overridesOnly);
    drawTunables();
  };

  function pollTunables(force) {
    if (!devOn) return;
    if (!force && Date.now() - tunTimer < 5000) return;
    tunTimer = Date.now();

    fetch('/api/tunables', { cache: 'no-store' })
      .then(function (r) { return r.json(); })
      .then(function (d) { tunables = d.tunables; drawTunables(); })
      .catch(function () {});
  }

  function drawTunables() {
    var q = document.getElementById('tunsearch').value.trim().toLowerCase();

    var rows = tunables.filter(function (t) {
      if (overridesOnly && !t.overridden) return false;
      if (!q) return true;
      return t.key.toLowerCase().indexOf(q) >= 0 || t.label.toLowerCase().indexOf(q) >= 0;
    });

    if (!rows.length) {
      document.getElementById('tunbox').innerHTML =
        '<span class="k">nothing matches. If the list is empty entirely, no element has registered its knobs yet: show the map or NavBall tag.</span>';
      return;
    }

    // Grouped by element (the key prefix) then by the section the dev window uses, so the web list reads in
    // the same order as the window on the headset. One mental model, two views of it.
    var groups = {};
    rows.forEach(function (t) {
      var g = t.key.split('.')[0] + '  /  ' + (t.section || 'General');
      (groups[g] = groups[g] || []).push(t);
    });

    var html = '';
    Object.keys(groups).sort().forEach(function (g) {
      html += '<div class="tsec">' + esc(g) + '</div>';
      groups[g].forEach(function (t) {
        var val = t.kind === 'bool' ? (t.value === '1' ? 'on' : 'off') : t.value;
        var def = t.overridden
          ? '<div class="tdef">was ' + esc(t.kind === 'bool' ? (t.default === '1' ? 'on' : 'off') : t.default) + '</div>'
          : '';
        html += '<div class="trow">' +
                  '<div><div class="tl">' + esc(t.label) + '</div><div class="tk">' + esc(t.key) + '</div></div>' +
                  '<div style="text-align:right"><span class="tval' + (t.overridden ? ' ovr' : '') + '">' +
                    esc(val) + '</span>' + def + '</div>' +
                '</div>';
      });
    });
    document.getElementById('tunbox').innerHTML = html;
  }

  function poll() {
    fetch('/api/status', { cache: 'no-store' })
      .then(function (r) { return r.json(); })
      .then(function (d) { lastOk = Date.now(); render(d); })
      .catch(function () { /* handled by the staleness check */ });

    pollFiles();

    // Only when DEV is open. With it closed the page costs exactly what it always did, so the run-day
    // dashboard is not paying for a log stream nobody is reading.
    if (devOn) {
      pollLogs();
      if (devPane === 'tun') pollTunables(false);
    }
  }

  // Listing recurses the trials directory ON THE HEADSET. Doing that twice a second, for every connected
  // browser, is real disk work for a list that changes once per trial. Every 5 s, and only redrawn when it
  // actually changed.
  var lastFilesJson = '';
  var filesTimer = 0;

  function pollFiles() {
    if (Date.now() - filesTimer < 5000) return;
    filesTimer = Date.now();

    fetch('/api/files', { cache: 'no-store' })
      .then(function (r) { return r.text(); })
      .then(function (text) {
        if (text === lastFilesJson) return;
        lastFilesJson = text;
        var d = JSON.parse(text);

        var ul = document.getElementById('files');
        if (!d.files.length) { ul.innerHTML = '<li class="k">none yet</li>'; return; }

        // Grouped by folder, and _incomplete is called out. A partial file from a crashed run sitting
        // undistinguished in a flat list is exactly how bad data gets analyzed as good.
        var groups = {};
        d.files.forEach(function (f) { (groups[f.folder || '(root)'] = groups[f.folder || '(root)'] || []).push(f); });

        var html = '';
        Object.keys(groups).sort().forEach(function (g) {
          var label = g === '_incomplete' ? '_incomplete (crashed or unfinished, NOT usable as-is)' : g;
          html += '<li><strong>' + label + '</strong><ul>' + groups[g].map(function (f) {
            return '<li><a href="/api/file?name=' + encodeURIComponent(f.path) + '">' + f.name + '</a>' +
                   ' <span class="k">(' + (f.bytes / 1024).toFixed(0) + ' KB)</span></li>';
          }).join('') + '</ul></li>';
        });
        ul.innerHTML = html;
      })
      .catch(function () {});
  }

  // Staleness is checked on the CLIENT clock, not the server's. A dashboard that silently keeps showing
  // the last good numbers after the headset drops is worse than one that shows nothing.
  setInterval(function () {
    var age = (Date.now() - lastOk) / 1000;
    var stale = lastOk > 0 && age > 3;
    document.getElementById('staleBox').classList.toggle('hidden', !stale);
    if (stale) txt('staleAge', age.toFixed(0));
    if (lastOk === 0) txt('conn', 'no response from the headset yet');
  }, 1000);

  // Phones freeze background JS timers when the screen locks or you switch apps. On return, lastOk is
  // minutes old and the staleness check would scream that the headset died when nothing happened at all.
  // A false alarm mid-session is expensive: it is the alarm you learn to ignore.
  document.addEventListener('visibilitychange', function () {
    if (document.visibilityState !== 'visible') return;
    lastOk = Date.now();                                  // do not judge the link on a timer that was frozen
    document.getElementById('staleBox').classList.add('hidden');
    filesTimer = 0;
    poll();                                               // re-establish the truth immediately

    // The browser silently drops the wake lock while hidden, so it has to be retaken or the screen starts
    // sleeping again halfway through the session with the button still cheerfully reading ON.
    if (wakeLock === null && document.getElementById('wake').textContent.indexOf('ON') >= 0) toggleWake();
  });

  window.poll = poll;   // so a button press can refresh immediately instead of waiting for the next tick

  // The text filters redraw from what is already held, so typing never refetches.
  document.getElementById('logsearch').addEventListener('input', drawLog);
  document.getElementById('tunsearch').addEventListener('input', drawTunables);

  applyDev();           // DEV survives a reload, so a debugging session is not undone by a refresh
  if (devOn) { pollLogs(); pollTunables(true); }

  poll();
  setInterval(poll, 500);   // 2 Hz. Nothing here needs to be faster; the data path is not this page.
})();
</script>
